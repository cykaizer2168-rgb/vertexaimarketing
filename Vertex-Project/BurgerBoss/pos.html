function doPost(e) {
  try {
    const data = JSON.parse(e && e.postData ? e.postData.contents : '{}');
    if (data.action === 'log_sale')     logSale(data);
    if (data.action === 'deduct_stock') deductStock(data.items || []);
    if (data.action === 'adjust_stock') adjustStock(data.items || [], data.reason || '', data.doneBy || 'POS');
    if (data.action === 'log_purchase')  logPurchase(data);
    if (data.action === 'log_eod')       logEOD(data);
    if (data.action === 'log_expense')   logExpense(data);
    if (data.action === 'add_menu_item')    addMenuItem(data);
    if (data.action === 'add_recipe_row')    addRecipeRow(data);
    if (data.action === 'delete_recipe_row') deleteRecipeRow(data);
    if (data.action === 'update_recipe_cost') updateRecipeCost(data);
    return json({ ok: true });
  } catch(err) {
    return json({ ok: false, error: err.message });
  }
}

function logPurchase(d) {
  const ss      = SpreadsheetApp.openById(SPREADSHEET_ID);
  const items   = d.items || [];
  const now     = new Date();
  const nowStr  = now.toLocaleString('en-PH');
  const dateStr = d.date || now.toLocaleDateString('en-PH', {year:'numeric',month:'2-digit',day:'2-digit'});
  const timeStr = now.toLocaleTimeString('en-PH', {hour:'2-digit',minute:'2-digit'});
  const doneBy  = d.doneBy || 'POS';

  // 1. Log to Purchases tab (create if not exists)
  const PUR_HEADERS = ['Date','Receipt #','Supplier','Item','Qty','Unit','Unit Cost','Line Total','Logged By','Timestamp'];
  let purSheet = ss.getSheetByName('Purchases');
  if (!purSheet) {
    purSheet = ss.insertSheet('Purchases');
    purSheet.appendRow(PUR_HEADERS);
    purSheet.getRange(1,1,1,PUR_HEADERS.length).setFontWeight('bold');
  }
  // Ensure all expected columns exist (add missing ones to the right)
  const existingHeaders = purSheet.getRange(1,1,1,purSheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  PUR_HEADERS.forEach(h => {
    if (!existingHeaders.includes(h)) {
      const newCol = purSheet.getLastColumn() + 1;
      purSheet.getRange(1, newCol).setValue(h).setFontWeight('bold');
      existingHeaders.push(h);
    }
  });
  items.forEach(item => {
    const dataMap = {
      'Date':       dateStr,
      'Receipt #':  d.receiptNo || '',
      'Supplier':   d.supplier  || '',
      'Item':       item.name,
      'Qty':        item.qty,
      'Unit':       item.unit || 'pcs',
      'Unit Cost':  item.unitCost  || 0,
      'Line Total': item.lineTotal || 0,
      'Logged By':  doneBy,
      'Timestamp':  now.toISOString()
    };
    purSheet.appendRow(existingHeaders.map(h => dataMap[h] !== undefined ? dataMap[h] : ''));
  });

  // 2. Update Inventory stock + log to stock adjustments
  const invSheet = ss.getSheetByName('Inventory');
  const logSheet = ss.getSheetByName('stock adjustments');
  if (!invSheet) return;

  const invData    = invSheet.getDataRange().getValues();
  const invHeaders = invData[0].map(h => String(h).trim());
  const itemCol    = invHeaders.indexOf('Item');
  const stockCol   = invHeaders.indexOf('Stock');
  const updatedCol = invHeaders.indexOf('Last Updated');
  const restockCol = invHeaders.indexOf('Last Restocked');
  const rstQtyCol  = invHeaders.indexOf('Last Restock Qty');

  items.forEach(item => {
    const qty = parseFloat(item.qty) || 0;
    for (let i = 1; i < invData.length; i++) {
      if (String(invData[i][itemCol]).trim() !== String(item.name).trim()) continue;
      const current  = parseFloat(invData[i][stockCol]) || 0;
      const newStock = current + qty;
      invSheet.getRange(i + 1, stockCol + 1).setValue(newStock);
      if (updatedCol >= 0) invSheet.getRange(i + 1, updatedCol + 1).setValue(nowStr);
      if (restockCol >= 0) invSheet.getRange(i + 1, restockCol + 1).setValue(nowStr);
      if (rstQtyCol  >= 0) invSheet.getRange(i + 1, rstQtyCol  + 1).setValue(qty);
      // Log to stock adjustments tab
      if (logSheet) {
        logSheet.appendRow([
          dateStr, timeStr, item.name, 'Item Receipt',
          qty, current, newStock,
          `Item Receipt${d.receiptNo ? ' - ' + d.receiptNo : ''}`,
          doneBy, now.toISOString()
        ]);
      }
      break;
    }
  });
}

function updateRecipeCost(d) {
  const ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('Recipes');
  if (!sheet || sheet.getLastRow() < 2) return;
  const [headers, ...rows] = sheet.getDataRange().getValues();
  const h      = headers.map(x => String(x).trim());
  const nmCol  = h.findIndex(k => /menu.?item.?name/i.test(k) || /item.?name/i.test(k));
  const ingCol = h.findIndex(k => /^ingredient/i.test(k));
  let   ucCol  = h.findIndex(k => /unit.?cost/i.test(k));
  if (ucCol < 0) {
    ucCol = sheet.getLastColumn();
    sheet.getRange(1, ucCol + 1).setValue('Unit Cost').setFontWeight('bold');
  }
  for (let i = 0; i < rows.length; i++) {
    if (String(rows[i][nmCol]).trim()  === String(d.menuItemName).trim() &&
        String(rows[i][ingCol]).trim() === String(d.ingredient).trim()) {
      sheet.getRange(i + 2, ucCol + 1).setValue(parseFloat(d.unitCost) || 0);
    }
  }
}

function deleteRecipeRow(d) {
  const ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('Recipes');
  if (!sheet || sheet.getLastRow() < 2) return;
  const [headers, ...rows] = sheet.getDataRange().getValues();
  const h    = headers.map(x => String(x).trim());
  const nmCol  = h.indexOf('Menu Item Name');
  const ingCol = h.indexOf('Ingredient');
  if (nmCol < 0 || ingCol < 0) return;
  for (let i = rows.length - 1; i >= 0; i--) {
    if (String(rows[i][nmCol]).trim()  === String(d.menuItemName).trim() &&
        String(rows[i][ingCol]).trim() === String(d.ingredient).trim()) {
      sheet.deleteRow(i + 2); // +1 for header, +1 for 1-indexed
    }
  }
}

  }

  const date    = d.date || '';
  const doneBy  = d.doneBy || 'POS';
  const ts      = d.timestamp || new Date().toISOString();

  // Log cash row
  const c = d.cash || {};
  sheet.appendRow([date,'Cash','—','₱', c.beginning||0, c.cashSales||0, c.purchases||0, c.ending||0, 'Cash Summary', doneBy, ts]);

  // Log stock rows
  (d.stock || []).forEach(r => {
    sheet.appendRow([date,'Stock', r.name, r.unit||'pcs', r.beginning||0, r.delivery||0, r.offtake||0, r.ending||0, '', doneBy, ts]);
  });
}

function adjustStock(items, reason, doneBy) {
  const ss       = SpreadsheetApp.openById(SPREADSHEET_ID);
  const invSheet = ss.getSheetByName('Inventory');
  if (!invSheet) return;

  const invData       = invSheet.getDataRange().getValues();
  const headers       = invData[0].map(h => String(h).trim());
  const itemCol       = headers.indexOf('Item');
  const stockCol      = headers.indexOf('Stock');
  const updatedCol    = headers.indexOf('Last Updated');
  const notesCol      = headers.indexOf('Notes');
  const restockCol    = headers.indexOf('Last Restocked');
  const restockQtyCol = headers.indexOf('Last Restock Qty');

  // Use the existing 'stock adjustments' log tab
  const logSheet  = ss.getSheetByName('stock adjustments');
  const doneByStr = doneBy || 'POS';

  const now     = new Date();
  const nowStr  = now.toLocaleString('en-PH');
  const dateStr = now.toLocaleDateString('en-PH', { year:'numeric', month:'2-digit', day:'2-digit' });
  const timeStr = now.toLocaleTimeString('en-PH', { hour:'2-digit', minute:'2-digit' });

  items.forEach(adj => {
    for (let i = 1; i < invData.length; i++) {
      if (String(invData[i][itemCol]).trim() !== String(adj.name).trim()) continue;
      const current  = parseFloat(invData[i][stockCol]) || 0;
      const delta    = parseFloat(adj.delta) || 0;
      const newStock = Math.max(0, current + delta);

      // Update Inventory sheet
      invSheet.getRange(i + 1, stockCol + 1).setValue(newStock);
      if (updatedCol >= 0)   invSheet.getRange(i + 1, updatedCol + 1).setValue(nowStr);
      if (notesCol >= 0)     invSheet.getRange(i + 1, notesCol + 1).setValue(reason || '');
      if (delta > 0) {
        if (restockCol >= 0)    invSheet.getRange(i + 1, restockCol + 1).setValue(nowStr);
        if (restockQtyCol >= 0) invSheet.getRange(i + 1, restockQtyCol + 1).setValue(delta);
      }

      // Append to existing 'stock adjustments' log tab
      // Columns: Date | Time | Item | Adjustment Type | Qty Changed | Old Stock | New Stock | Reason | Done By | Timestamp
      if (logSheet) {
        logSheet.appendRow([
          dateStr,
          timeStr,
          adj.name,
          reason || '',
          delta,
          current,
          newStock,
          reason || '',
          doneByStr,
          now.toISOString()
        ]);
      }
      break;
    }
  });
}

function deductStock(orderItems) {
  const ss       = SpreadsheetApp.openById(SPREADSHEET_ID);
  const invSheet = ss.getSheetByName('Inventory');
  if (!invSheet) return;

  const invData    = invSheet.getDataRange().getValues();
  const invHeaders = invData[0].map(h => String(h).trim());
  const itemCol    = invHeaders.indexOf('Item');
  const stockCol   = invHeaders.indexOf('Stock');
  const updatedCol = invHeaders.indexOf('Last Updated');
  const now        = new Date().toLocaleString('en-PH');

  // Build deduction map: { ingredientName: totalQty }
  const deduct = {};
  const recSheet = ss.getSheetByName('Recipes');

  if (recSheet && recSheet.getLastRow() > 1) {
    // ── BOM mode: deduct ingredients based on Recipes tab ──
    const [recHeaders, ...recRows] = recSheet.getDataRange().getValues();
    const rh      = recHeaders.map(h => String(h).trim());
    const rIdCol  = rh.indexOf('Menu Item ID');
    const rNmCol  = rh.indexOf('Menu Item Name');
    const rIngCol = rh.indexOf('Ingredient');
    const rQtyCol = rh.indexOf('Qty Used');
    orderItems.forEach(item => {
      recRows
        .filter(r => {
          const matchId   = rIdCol  >= 0 && String(r[rIdCol]).trim()  === String(item.id  || '').trim();
          const matchName = rNmCol  >= 0 && String(r[rNmCol]).trim().toLowerCase() === String(item.name || '').trim().toLowerCase();
          return matchId || matchName;
        })
        .forEach(r => {
          const ing = String(r[rIngCol]).trim().toLowerCase();
          const qty = (parseFloat(r[rQtyCol]) || 0) * (item.qty || 1);
          deduct[ing] = (deduct[ing] || 0) + qty;
        });
    });
  } else {
    // ── Direct mode: match menu item name to Inventory item name (case-insensitive) ──
    orderItems.forEach(item => {
      const name = String(item.name || '').trim().toLowerCase();
      if (name) deduct[name] = (deduct[name] || 0) + (item.qty || 1);
    });
  }

  // Apply deductions to Inventory sheet (case-insensitive name match)
  for (let i = 1; i < invData.length; i++) {
    const name = String(invData[i][itemCol]).trim().toLowerCase();
    if (deduct[name] === undefined) continue;
    const current  = parseFloat(invData[i][stockCol]) || 0;
    const newStock = Math.max(0, current - deduct[name]);
    invSheet.getRange(i + 1, stockCol + 1).setValue(newStock);
    if (updatedCol >= 0) invSheet.getRange(i + 1, updatedCol + 1).setValue(now);
  }
}
